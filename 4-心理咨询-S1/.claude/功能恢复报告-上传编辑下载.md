```javascript
function uploadApproachWord(approachName, visitorId, visitId) {
    // 1. 创建文件输入
    const fileInput = document.createElement('input');
    fileInput.type = 'file';
    fileInput.accept = '.docx';
    
    // 2. 文件选择后处理
    fileInput.onchange = async (e) => {
        const file = e.target.files[0];
        const arrayBuffer = await file.arrayBuffer();
        
        // 3. 使用mammoth.js解析
        const result = await mammoth.convertToHtml({arrayBuffer: arrayBuffer});
        
        // 4. 填充到编辑区
        const contentDiv = document.getElementById('analysis-content-' + approachName);
        contentDiv.innerHTML = result.value;
        
        // 5. 自动进入编辑模式
        toggleEdit(approachName);
        
        alert('Word文档已上传，请点击"保存"按钮保存内容');
    };
    
    fileInput.click();
}
```

**依赖库**:
- mammoth.js - Word文档解析

#### 4. 编辑功能详解 ✅

**编辑模式切换**:
```javascript
function toggleEdit(approachName) {
    const contentDiv = document.getElementById('analysis-content-' + approachName);
    const editBtn = document.getElementById('edit-btn-' + approachName);
    const saveBtn = document.getElementById('save-btn-' + approachName);
    const toolbar = document.getElementById('toolbar-' + approachName);
    
    if (contentDiv.getAttribute('contenteditable') === 'true') {
        // 退出编辑模式
        contentDiv.setAttribute('contenteditable', 'false');
        editBtn.style.display = 'inline-block';
        saveBtn.style.display = 'none';
        toolbar.style.display = 'none';
    } else {
        // 进入编辑模式
        contentDiv.setAttribute('contenteditable', 'true');
        contentDiv.focus();
        editBtn.style.display = 'none';
        saveBtn.style.display = 'inline-block';
        toolbar.style.display = 'block';
        currentEditingApproach = approachName;
    }
}
```

**保存功能**:
```javascript
async function saveEdit(approachName, visitorId, visitId) {
    const contentDiv = document.getElementById('analysis-content-' + approachName);
    const content = contentDiv.innerHTML;
    
    // 构建更新数据
    const updateData = {
        visitor_id: visitorId,
        visit_id: visitId,
        approach: approachName,
        content: content
    };
    
    // 发送到API
    const response = await fetch('http://localhost:5001/api/update-approach-analysis', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify(updateData)
    });
    
    if (response.ok) {
        alert('保存成功！');
        toggleEdit(approachName);
    } else {
        alert('保存失败，请检查服务器');
    }
}
```
